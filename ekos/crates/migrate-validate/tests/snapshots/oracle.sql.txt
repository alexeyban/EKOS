Text
  (CASE WHEN "txt" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("txt"), '\', '\\'), CHR(31), '\x1f') END)
Char
  (CASE WHEN "ch" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("ch"), '\', '\\'), CHR(31), '\x1f') END)
Bool
  (CASE WHEN "flag" IS NULL THEN '\N' ELSE (CASE WHEN "flag" = 1 THEN 't' ELSE 'f' END) END)
Int
  (CASE WHEN "n" IS NULL THEN '\N' ELSE TO_CHAR("n") END)
Decimal(2)
  (CASE WHEN "amt" IS NULL THEN '\N' ELSE (CASE WHEN "amt" <> ROUND("amt", 2) THEN TO_CHAR("amt") ELSE TO_CHAR("amt", 'FM999999999999999999999999999999999990.00') END) END)
TimestampUtc
  (CASE WHEN "ts" IS NULL THEN '\N' ELSE TO_CHAR(CAST(SYS_EXTRACT_UTC("ts") AS TIMESTAMP(6)), 'YYYY-MM-DD"T"HH24:MI:SS.FF6') END)
TimestampNaive
  (CASE WHEN "ts_naive" IS NULL THEN '\N' ELSE TO_CHAR(CAST("ts_naive" AS TIMESTAMP(6)), 'YYYY-MM-DD"T"HH24:MI:SS.FF6') END)
Date
  (CASE WHEN "d" IS NULL THEN '\N' ELSE TO_CHAR("d", 'YYYY-MM-DD') END)
Time
  (CASE WHEN "t" IS NULL THEN '\N' ELSE TO_CHAR(CAST("t" AS TIMESTAMP(6)), 'HH24:MI:SS.FF6') END)
Uuid
  (CASE WHEN "u" IS NULL THEN '\N' ELSE LOWER(TO_CHAR("u")) END)
Bytes
  (CASE WHEN "b" IS NULL THEN '\N' ELSE LOWER(RAWTOHEX("b")) END)
Inet
  (CASE WHEN "ip" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("ip"), '\', '\\'), CHR(31), '\x1f') END)
ROW_HASH
  LOWER(RAWTOHEX(STANDARD_HASH((CASE WHEN "txt" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("txt"), '\', '\\'), CHR(31), '\x1f') END) || CHR(31) || (CASE WHEN "ch" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("ch"), '\', '\\'), CHR(31), '\x1f') END), 'SHA256')))
BUCKET_QUERY
  SELECT bucket, COUNT(*) AS row_count, SUM(h) AS hash_sum FROM (SELECT MOD(TO_NUMBER(SUBSTR(LOWER(RAWTOHEX(STANDARD_HASH((CASE WHEN "id" IS NULL THEN '\N' ELSE TO_CHAR("id") END), 'SHA256'))), 1, 15), 'XXXXXXXXXXXXXXX'), 4096) AS bucket, TO_NUMBER(SUBSTR(LOWER(RAWTOHEX(STANDARD_HASH((CASE WHEN "txt" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("txt"), '\', '\\'), CHR(31), '\x1f') END) || CHR(31) || (CASE WHEN "ch" IS NULL THEN '\N' ELSE REPLACE(REPLACE(TO_CHAR("ch"), '\', '\\'), CHR(31), '\x1f') END), 'SHA256'))), 1, 15), 'XXXXXXXXXXXXXXX') AS h FROM s.orders) GROUP BY bucket ORDER BY bucket
