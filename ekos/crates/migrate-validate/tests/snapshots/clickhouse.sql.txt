Text
  if(isNull("txt"), '\N', replaceAll(replaceAll(toString("txt"), '\\', '\\\\'), char(31), '\\x1f'))
Char
  if(isNull("ch"), '\N', replaceAll(replaceAll(toString("ch"), '\\', '\\\\'), char(31), '\\x1f'))
Bool
  if(isNull("flag"), '\N', if("flag", 't', 'f'))
Int
  if(isNull("n"), '\N', toString("n"))
Decimal(2)
  if(isNull("amt"), '\N', toString(toDecimal128("amt", 2)))
TimestampUtc
  if(isNull("ts"), '\N', formatDateTime(toTimeZone("ts", 'UTC'), '%Y-%m-%dT%H:%i:%S.%f'))
TimestampNaive
  if(isNull("ts_naive"), '\N', formatDateTime("ts_naive", '%Y-%m-%dT%H:%i:%S.%f'))
Date
  if(isNull("d"), '\N', formatDateTime("d", '%Y-%m-%d'))
Time
  if(isNull("t"), '\N', formatDateTime("t", '%H:%i:%S.%f'))
Uuid
  if(isNull("u"), '\N', lower(toString("u")))
Bytes
  if(isNull("b"), '\N', lower(hex("b")))
Inet
  if(isNull("ip"), '\N', replaceAll(replaceAll(toString("ip"), '\\', '\\\\'), char(31), '\\x1f'))
ROW_HASH
  lower(hex(MD5(if(isNull("txt"), '\N', replaceAll(replaceAll(toString("txt"), '\\', '\\\\'), char(31), '\\x1f')) || char(31) || if(isNull("ch"), '\N', replaceAll(replaceAll(toString("ch"), '\\', '\\\\'), char(31), '\\x1f')))))
BUCKET_QUERY
  SELECT reinterpretAsUInt64(reverse(unhex(concat('0', substring(lower(hex(MD5(if(isNull("id"), '\N', toString("id"))))), 1, 15))))) % 4096 AS bucket, count(*) AS row_count, sum(toDecimal128(reinterpretAsUInt64(reverse(unhex(concat('0', substring(lower(hex(MD5(if(isNull("txt"), '\N', replaceAll(replaceAll(toString("txt"), '\\', '\\\\'), char(31), '\\x1f')) || char(31) || if(isNull("ch"), '\N', replaceAll(replaceAll(toString("ch"), '\\', '\\\\'), char(31), '\\x1f'))))), 1, 15))))), 0)) AS hash_sum FROM s.orders GROUP BY bucket ORDER BY bucket
